  BackgroundColor #F7F9FC
  BorderColor #555555

package "Capa cliente" {
  package "Cliente Web (Angular)" {
    [UIVentas] as UIVEN
    [UIEntregas] as UIENT
  }
  package "Cliente de escritorio (C++)" {
    [UIRecursosHumanos] as UIRH
    [UIBodega] as UIBOD
  }
}

package "Capa de negocio" {
  () "IPoliMarket\n(API REST)" as IPM
  [FachadaPoliMarket\n<<Facade>>] as FAC
  IPM - FAC

  package "Componentes de área" {
    () "IGestionPersonal" as iGP
    () "IAutorizacion" as iAUT
    [CompRecursosHumanos] as RH
    iGP -- RH
    iAUT -- RH

    () "IClientes" as iCLI
    () "IVentas" as iVEN
    [CompVentas] as VEN
    iCLI -- VEN
    iVEN -- VEN

    () "IEntregas" as iENT
    [CompEntregas] as ENT
    iENT -- ENT

    () "IInventario" as iINV
    [CompBodega] as BOD
    iINV -- BOD

    () "IProveedores" as iPRO
    [CompProveedores] as PRO
    iPRO -- PRO
  }

  package "Componentes transversales" {
    () "IAutenticacion" as iAUTN
    [CompSeguridad] as SEG
    iAUTN -- SEG
    () "ITerceros" as iTER
    [CompTerceros] as TER
    iTER -- TER
  }
}

UIRH ..> IPM
